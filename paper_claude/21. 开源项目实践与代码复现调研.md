#### 🔥 最难写的代码逻辑

**1. 双阶段训练状态管理**

```python
# 难点：在两个训练阶段之间正确切换和维护模型状态
class WakeSleepTrainer:
    def __init__(self, high_level_policy, low_level_policy):
        self.phase = 'sleep'  # 或 'wake'
        self.high_policy = high_level_policy
        self.low_policy = low_level_policy

        # 关键：需要维护两套参数状态
        self.frozen_high_params = None

    def set_sleep_phase(self):
        """进入睡眠阶段 - 冻结高层"""
        self.phase = 'sleep'
        self.frozen_high_params = copy.deepcopy(self.high_policy.state_dict())

        # 关键：将高层替换为固定策略（平均预算分配）
        self.high_policy.eval()  # 冻结参数
        for param in self.high_policy.parameters():
            param.requires_grad = False

    def set_wake_phase(self):
        """进入清醒阶段 - 联合训练"""
        self.phase = 'wake'

        # 恢复高层参数的可训练性
        for param in self.high_policy.parameters():
            param.requires_grad = True
        self.high_policy.train()

        # 但保持之前sleep阶段学到的权重
```

**为什么难？**
- 需要精确控制何时冻结/解冻参数
- 必须正确保存和恢复模型状态
- 两个阶段使用不同的更新方法（MC vs TD）

**2. MC和TD混合更新逻辑**

```python
# 难点：高层使用MC方法，低层使用TD方法，需要两套不同的更新管道
def update_policies(self, trajectory):
    """根据训练阶段选择不同的更新方法"""

    if self.phase == 'sleep':
        # 睡眠阶段：只更新低层，使用TD方法
        # 高层使用固定策略，不更新

        # 低层TD更新
        for t in range(len(trajectory) - 1):
            s_t, a_t, r_t, s_t1 = trajectory[t]

            # TD目标：r_t + γ * Q(s_{t+1}, a_{t+1})
            td_target = r_t + self.gamma * self.low_q[s_t1].max()

            # TD更新：Q(s,a) ← Q(s,a) + α[TD_target - Q(s,a)]
            td_error = td_target - self.low_q[s_t, a_t]
            self.low_q[s_t, a_t] += self.alpha_low * td_error

        # 高层离线预训练（使用MC方法但不在线更新）
        self._pretrain_high_offline(trajectory)

    else:  # wake phase
        # 清醒阶段：两层都更新
        # 高层用MC，低层用TD

        # 高层MC更新
        G = 0  # 回报
        for t in reversed(range(len(trajectory))):
            s_t, o_t = trajectory[t]['state'], trajectory[t]['option']
            r_t = trajectory[t]['reward']
            G = r_t + self.gamma * G  # 累计回报

            # MC更新：Q(s,o) ← Q(s,o) + α[G - Q(s,o)]
            mc_error = G - self.high_q[s_t, o_t]
            self.high_q[s_t, o_t] += self.alpha_high * mc_error

        # 低层继续TD更新
        # ... (同sleep阶段)
```

**为什么难？**
- MC和TD有不同的更新时机和公式
- 需要同时维护两种经验存储
- 容易混淆两种方法的梯度计算

```python
    def __init__(self):
        self.sleep_buffer = []
        self.wake_buffer = []

    def add(self, experience, phase):
        if phase == 'sleep':
            self.sleep_buffer.append(experience)
        else:
            self.wake_buffer.append(experience)

    def sample(self, batch_size, phase):
        buffer = self.sleep_buffer if phase == 'sleep' else self.wake_buffer
        return random.sample(buffer, min(batch_size, len(buffer)))
```

### 2.2 论文B - HEM的两层交互逻辑复现难点

#### 🔥 最难写的代码逻辑

**1. 层次化策略梯度的计算**

```python
# 难点：正确分离高层和低层的梯度，避免相互干扰
class HierarchicalPolicyGradient:
    def compute_gradients(self, batch):
        """计算层次策略梯度"""

        # 1. 前向传播：采样轨迹
        log_probs_high = []
        log_probs_low = []
        rewards = []

        for state in batch:
            # 高层：采样k
            k_dist = self.high_policy(state)
            k = k_dist.sample()
            log_prob_high = k_dist.log_prob(k)

            # 低层：给定k，采样具体割
            cut_dist = self.low_policy(state, k)
            cuts = cut_dist.sample()
            log_prob_low = cut_dist.log_prob(cuts)

            # 执行并获取奖励
            reward = self.env.execute_cuts(cuts)

            log_probs_high.append(log_prob_high)
            log_probs_low.append(log_prob_low)
            rewards.append(reward)

        # 2. 计算优势（使用Critic或简单baseline）
        advantages = self._compute_advantages(rewards)

        # 3. 关键：分别计算两个梯度
        # 高层梯度：使用条件期望奖励 E[r|k]
        high_gradient = self._compute_high_gradient(
            log_probs_high,
            advantages,
            k_values  # 需要按k分组计算平均奖励
        )

        # 低层梯度：使用实际奖励 r
        low_gradient = self._compute_low_gradient(
            log_probs_low,
            advantages  # 直接使用实际奖励
        )

        return high_gradient, low_gradient

    def _compute_high_gradient(self, log_probs, advantages, k_values):
        """
        高层梯度计算的关键：
        - 需要对相同的k值计算平均奖励
        - 梯度公式：∇θ₁ J = E[∇log P(k|s) × E[r|s,k]]
        """
        # 按k分组
        k_groups = {}
        for k, log_prob, adv in zip(k_values, log_probs, advantages):
            if k not in k_groups:
                k_groups[k] = {'log_probs': [], 'rewards': []}
            k_groups[k]['log_probs'].append(log_prob)
            k_groups[k]['rewards'].append(adv)

        # 计算每组k的平均奖励
        gradients = []
        for k, group in k_groups.items():
            avg_reward = np.mean(group['rewards'])
            for log_prob in group['log_probs']:
                # 高层只对k的选择负责
                gradients.append(-log_prob * avg_reward)

        return torch.stack(gradients).mean()
```

**为什么难？**
- 需要正确实现论文中的层次梯度公式
- 高层和低层的梯度计算方式不同
- 梯度容易混淆或相互干扰

**2. Pointer Network的掩码机制**

```python
# 难点：正确实现防止重复选择的掩码机制
class PointerNetworkDecoder(nn.Module):
    def forward(self, node_embeddings, k, training=True):
        """
        节点选择器：逐步选择k个不同的节点

        Args:
            node_embeddings: (batch, N, embed_dim) 所有节点的嵌入
            k: 要选择的节点数量
            training: 是否训练模式（训练时采样，推理时取argmax）
        """
        batch_size, N, embed_dim = node_embeddings.shape
        device = node_embeddings.device

        # 初始化
        mask = torch.ones(batch_size, N, device=device)  # 可选掩码
        selected_nodes = []
        log_probs = []

        hidden = self.init_hidden(batch_size, device)

        for step in range(k):
            # 计算注意力分数
            query = self.W_query(hidden)  # (batch, embed_dim)

            # 点积注意力
            logits = torch.bmm(
                node_embeddings,  # (batch, N, embed_dim)
                query.unsqueeze(-1)  # (batch, embed_dim, 1)
            ).squeeze(-1)  # (batch, N)

            logits = logits / math.sqrt(embed_dim)  # 缩放

            # ⚠️ 关键：应用掩码防止重复选择
            logits = logits.masked_fill(mask == 0, float('-inf'))

            # Softmax得到概率分布
            probs = F.softmax(logits, dim=-1)  # (batch, N)

            # 选择节点
            if training:
                selected = torch.multinomial(probs, 1).squeeze(-1)
            else:
                selected = torch.argmax(probs, dim=-1)

            # 记录
            selected_nodes.append(selected)
            log_probs.append(F.log_softmax(logits, dim=-1))

            # 更新掩码：已选节点设为0
            mask[torch.arange(batch_size), selected] = 0

            # 更新hidden状态（包含已选节点信息）
            selected_embed = node_embeddings[torch.arange(batch_size), selected]
            hidden = self.gru_cell(selected_embed, hidden)

        return torch.stack(selected_nodes, dim=1), torch.stack(log_probs, dim=1)
```

**为什么难？**
- 掩码必须在每步正确更新
- batch中每个样本的掩码独立
- 需要处理训练和推理的不同行为

```python
# ============================================================================
# WS-option Wake-sleep Training Algorithm
# ============================================================================

def ws_option_train(env, high_policy, low_policy, config):
    """
    WS-option算法的Wake-sleep训练主循环

    参数:
        env: 环境（返回奖励和成本）
        high_policy: 高层策略（option策略）
        low_policy: 低层策略（intra-option策略）
        config: 超参数配置
    """

    # 初始化
    beta = config['beta']  # 安全权重系数
    phase = 'sleep'  # 初始阶段

    # Sleep阶段配置
    sleep_iterations = config['sleep_iterations']
    wake_iterations = config['wake_iterations']

    # 使用平均策略（固定预算分配）
    avg_budget_strategy = AverageBudgetAllocator()

    # ========================================================================
    # 阶段1: Sleep阶段 - 冻结高层，训练低层
    # ========================================================================

    print("=== SLEEP PHASE: Freezing high-level, training low-level ===")

    # 冻结高层策略
    high_policy.freeze()

    for iteration in range(sleep_iterations):
        # 1. 使用平均预算分配策略收集轨迹
        trajectory = collect_trajectory_with_fixed_policy(
            env=env,
            high_policy=avg_budget_strategy,  # 固定策略
            low_policy=low_policy,
            beta=beta
        )

        # 2. 使用TD方法更新低层Q函数
        update_low_level_td(trajectory, low_policy)

        # 3. 离线预训练高层（在后台）
        pretrain_high_offline(trajectory, high_policy)

        if iteration % 100 == 0:
            print(f"Sleep iteration {iteration}, reward: {trajectory['total_reward']}")

    # ========================================================================
    # 阶段2: Wake阶段 - 联合训练两层
    # ========================================================================

    print("=== WAKE PHASE: Joint training of both levels ===")

    # 解冻高层策略
    high_policy.unfreeze()

    for iteration in range(wake_iterations):
        # 1. 使用当前策略收集轨迹
        trajectory = collect_trajectory_joint(
            env=env,
            high_policy=high_policy,
            low_policy=low_policy,
            beta=beta
        )

        # 2. 使用MC方法更新高层Q函数
        update_high_level_mc(trajectory, high_policy)

        # 3. 使用TD方法更新低层Q函数
        update_low_level_td(trajectory, low_policy)

        if iteration % 100 == 0:
            print(f"Wake iteration {iteration}, reward: {trajectory['total_reward']}")

    return high_policy, low_policy

# 辅助函数：轨迹收集
# ============================================================================

def collect_trajectory_with_fixed_policy(env, high_policy, low_policy, beta):
    """
    使用固定高层策略收集轨迹（Sleep阶段）
    """
    trajectory = {
        'states': [],
        'options': [],
        'actions': [],
        'rewards': [],
        'costs': [],
        'modified_rewards': []
    }

    state = env.reset()
    done = False

    while not done:
        # 高层：使用固定策略分配预算
        option = high_policy.select_option(state)  # 例如：平均分配

        # 低层：在给定预算下选择动作
        action = low_policy.select_action(state, option)

        # 环境执行
        next_state, reward, cost, done = env.step(action)

        # 关键：计算修正奖励
        modified_reward = reward - beta * cost

        # 存储
        trajectory['states'].append(state)
        trajectory['options'].append(option)
        trajectory['actions'].append(action)
        trajectory['rewards'].append(reward)
        trajectory['costs'].append(cost)
        trajectory['modified_rewards'].append(modified_reward)

        state = next_state

    trajectory['total_reward'] = sum(trajectory['modified_rewards'])
    return trajectory

# ============================================================================
# 更新函数：MC和TD
# ============================================================================

def update_high_level_mc(trajectory, high_policy):
    """
    使用Monte Carlo方法更新高层Q函数

    MC特点：使用完整回报，无偏但高方差
    """
    # 计算完整回报（从每个时间步到结束）
    G = 0
    returns = []

    for t in reversed(range(len(trajectory['states']))):
        reward = trajectory['modified_rewards'][t]
        G = reward + high_policy.gamma * G
        returns.append(G)

    returns.reverse()  # 恢复时间顺序

    # MC更新：Q(s,o) ← Q(s,o) + α[G - Q(s,o)]
    for t, state in enumerate(trajectory['states']):
        option = trajectory['options'][t]
        G = returns[t]

        current_Q = high_policy.get_Q(state, option)

        # MC更新
        high_policy.set_Q(
            state,
            option,
            current_Q + high_policy.alpha * (G - current_Q)
        )


def update_low_level_td(trajectory, low_policy):
    """
    使用Temporal Difference方法更新低层Q函数

    TD特点：使用bootstrapping，有偏但低方差
    """
    for t in range(len(trajectory['states']) - 1):
        state = trajectory['states'][t]
        option = trajectory['options'][t]
        action = trajectory['actions'][t]

        reward = trajectory['modified_rewards'][t]
        next_state = trajectory['states'][t + 1]
        next_option = trajectory['options'][t + 1]

        # 当前Q值
        current_Q = low_policy.get_Q(state, option, action)

        # TD目标：r + γ * max_a' Q(s', o', a')
        next_Q_max = low_policy.get_max_Q(next_state, next_option)
        td_target = reward + low_policy.gamma * next_Q_max

        # TD更新：Q(s,o,a) ← Q(s,o,a) + α[TD_target - Q(s,o,a)]
        td_error = td_target - current_Q
        low_policy.set_Q(
            state,
            option,
            action,
            current_Q + low_policy.alpha * td_error
        )
```

```python
# ============================================================================
# HEM (Hierarchical Sequence Model) Training Algorithm
# ============================================================================

def hem_train(milp_env, high_policy, low_policy, critic, config):
    """
    HEM算法的层次策略梯度训练

    参数:
        milp_env: MILP求解环境
        high_policy: 高层策略（预测选择多少个割，k）
        low_policy: 低层策略（Pointer Network，选择具体割）
        critic: 价值函数网络
        config: 超参数配置
    """

    optimizer_high = torch.optim.Adam(high_policy.parameters(), lr=config['lr_high'])
    optimizer_low = torch.optim.Adam(low_policy.parameters(), lr=config['lr_low'])
    optimizer_critic = torch.optim.Adam(critic.parameters(), lr=config['lr_critic'])

    for iteration in range(config['max_iterations']):

        # ====================================================================
        # 1. 收集一个batch的轨迹
        # ====================================================================

        batch_trajectories = []
        batch_states = []
        batch_ks = []
        batch_cut_sequences = []
        batch_rewards = []

        for _ in range(config['batch_size']):
            # 获取MILP实例状态
            state = milp_env.reset()

            # 高层：预测选择多少个割（k）
            k_dist = high_policy(state)  # 输出k的分布
            k = k_dist.sample()  # 采样
            log_prob_k = k_dist.log_prob(k)

            # 低层：给定k，选择具体的割序列
            cuts_dist = low_policy(state, k)  # Pointer Network
            cuts = cuts_dist.sample()  # 采样k个割
            log_prob_cuts = cuts_dist.log_prob(cuts)

            # 执行：将选择的割提交给MILP求解器
            next_state, reward, done = milp_env.step(cuts)

            batch_trajectories.append({
                'state': state,
                'k': k,
                'cuts': cuts,
                'reward': reward,
                'log_prob_k': log_prob_k,
                'log_prob_cuts': log_prob_cuts
            })

            batch_states.append(state)
            batch_ks.append(k)
            batch_cut_sequences.append(cuts)
            batch_rewards.append(reward)

        # ====================================================================
        # 2. 计算优势函数
        # ====================================================================

        # 将states转换为batch tensor
        states_tensor = torch.stack(batch_states)

        # 使用Critic估计状态价值
        state_values = critic(states_tensor).squeeze()

        # 计算优势：A(s, k, cuts) = R - V(s)
        rewards_tensor = torch.tensor(batch_rewards, dtype=torch.float32)
        advantages = rewards_tensor - state_values.detach()

        # 归一化优势（稳定训练）
        advantages = (advantages - advantages.mean()) / (advantages.std() + 1e-8)

        # ====================================================================
        # 3. 计算层次策略梯度（关键！）
        # ====================================================================

        # --------------------------
        # 高层梯度计算
        # --------------------------

        # 关键：高层使用条件期望奖励 E[r | k]
        # 需要按k值分组，计算每个k的平均奖励

        k_groups = {}
        for traj in batch_trajectories:
            k = traj['k'].item()
            if k not in k_groups:
                k_groups[k] = {
                    'log_probs': [],
                    'rewards': []
                }
            k_groups[k]['log_probs'].append(traj['log_prob_k'])
            k_groups[k]['rewards'].append(traj['reward'])

        # 计算每个k的平均奖励
        loss_high = 0
        for k, group in k_groups.items():
            avg_reward_k = sum(group['rewards']) / len(group['rewards'])

            for log_prob_k in group['log_probs']:
                # 高层损失：-log P(k|s) × E[r|s,k]
                # 注意：使用平均奖励，而不是单个轨迹的奖励
                loss_high += -log_prob_k * avg_reward_k

        loss_high = loss_high / len(batch_trajectories)

        # --------------------------
        # 低层梯度计算
        # --------------------------

        # 关键：低层使用实际奖励 r(s, cuts)
        loss_low = 0
        for traj, advantage in zip(batch_trajectories, advantages):
            # 低层损失：-log P(cuts|s,k) × r(s, cuts)
            # 注意：使用实际奖励（或优势），不是平均奖励
            loss_low += -traj['log_prob_cuts'] * advantage

        loss_low = loss_low / len(batch_trajectories)

        # ====================================================================
        # 4. Critic损失（MSE）
        # ====================================================================

        loss_critic = F.mse_loss(state_values.squeeze(), rewards_tensor)

        # ====================================================================
        # 5. 反向传播和参数更新
        # ====================================================================

        # 更新高层
        optimizer_high.zero_grad()
        loss_high.backward()
        torch.nn.utils.clip_grad_norm_(high_policy.parameters(), max_norm=0.5)
        optimizer_high.step()

        # 更新低层
        optimizer_low.zero_grad()
        loss_low.backward()
        torch.nn.utils.clip_grad_norm_(low_policy.parameters(), max_norm=0.5)
        optimizer_low.step()

        # 更新Critic
        optimizer_critic.zero_grad()
        loss_critic.backward()
        optimizer_critic.step()

        # ====================================================================
        # 6. 日志和评估
        # ====================================================================

        if iteration % config['log_interval'] == 0:
            avg_reward = sum(batch_rewards) / len(batch_rewards)
            print(f"Iteration {iteration}:")
            print(f"  Avg Reward: {avg_reward:.2f}")
            print(f"  High Loss: {loss_high.item():.4f}")
            print(f"  Low Loss: {loss_low.item():.4f}")
            print(f"  Critic Loss: {loss_critic.item():.4f}")

    return high_policy, low_policy, critic


# ============================================================================
# 低层策略：Pointer Network实现
# ============================================================================

class PointerNetworkLowLevel(nn.Module):
    """
    Pointer Network：用于选择k个割并排序

    关键机制：
    1. 注意力机制：计算当前状态与所有候选割的匹配度
    2. 掩码机制：防止重复选择同一个割
    3. 顺序决策：逐步选择k个割
    """

    def __init__(self, input_dim, embed_dim, hidden_dim):
        super().__init__()

        # 节点（候选割）编码器
        self.cutter_encoder = nn.Linear(input_dim, embed_dim)

        # 解码器（用于生成序列）
        self.decoder = nn.GRUCell(embed_dim, hidden_dim)

        # 注意力查询变换
        self.W_query = nn.Linear(hidden_dim, embed_dim)

    def forward(self, state, k):
        """
        Args:
            state: (batch, N, input_dim) N个候选割的特征
            k: (batch,) 或标量，要选择的割数量

        Returns:
            cuts: (batch, k) 选中的割索引
            log_probs: (batch, k, N) 每步的对数概率
        """
        batch_size, N, input_dim = state.shape

        # 编码所有候选割
        cutter_embeddings = self.cutter_encoder(state)  # (batch, N, embed_dim)

        # 初始化解码器隐藏状态
        hidden = torch.zeros(batch_size, self.hidden_dim, device=state.device)

        # 初始化掩码（所有割都可选）
        mask = torch.ones(batch_size, N, device=state.device)

        selected_cuts = []
        log_probs_list = []

        for step in range(k):
            # 更新解码器
            if step == 0:
                context = torch.zeros(batch_size, self.embed_dim, device=state.device)
            else:
                # 使用上一个选中的割的嵌入作为上下文
                context = cutter_embeddings[torch.arange(batch_size), selected_cuts[-1]]

            hidden = self.decoder(context, hidden)

            # 计算注意力分数
            query = self.W_query(hidden)  # (batch, embed_dim)

            # 点积注意力
            logits = torch.bmm(
                cutter_embeddings,  # (batch, N, embed_dim)
                query.unsqueeze(-1)  # (batch, embed_dim, 1)
            ).squeeze(-1)  # (batch, N)

            logits = logits / math.sqrt(self.embed_dim)  # 缩放

            # 关键：应用掩码，防止重复选择
            logits = logits.masked_fill(mask == 0, float('-inf'))

            # Softmax得到概率分布
            probs = F.softmax(logits, dim=-1)  # (batch, N)
            log_probs = F.log_softmax(logits, dim=-1)

            # 采样或选择
            if self.training:
                selected = torch.multinomial(probs, 1).squeeze(-1)
            else:
                selected = torch.argmax(probs, dim=-1)

            selected_cuts.append(selected)
            log_probs_list.append(log_probs)

            # 更新掩码：已选的割设为0
            mask[torch.arange(batch_size), selected] = 0

        return torch.stack(selected_cuts, dim=1), torch.stack(log_probs_list, dim=1)
```
